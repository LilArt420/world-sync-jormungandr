// Gives one intermediate Rgan a job.
// callfunc "F_WSJ_Job", "<name>", <bit>, <quest>;
function	script	F_WSJ_Job	{
	.@name$ = getarg(0);
	.@bit = getarg(1);
	if (wsj_ep20 != 240 || isbegin_quest(8846) != 1) {
		mes "[" + .@name$ + "]";
		mes "Rgan rgan. Work is good.";
		close;
	}
	if (wsj_ep20b & .@bit) {
		mes "[" + .@name$ + "]";
		mes "I know where I'm going! Thank you, thank you.";
		close;
	}
	mes "[" + .@name$ + "]";
	mes "A job? For me? Where should I go?";
	next;
	// Places already handed out are taken (bits 4096..65536 in wsj_ep20c).
	setarray .@place$, "The puddle by the west wall", "The old tower", "The rubbish dump (2nd floor)", "The junk heap (2nd floor)", "The storehouse (2nd floor)";
	for (.@i = 0; .@i < 5; .@i++) {
		if (wsj_ep20c & (1 << .@i))
			.@opt$[.@i] = .@place$[.@i] + " (taken)";
		else
			.@opt$[.@i] = .@place$[.@i];
	}
	.@s = select(implode(.@opt$, ":")) - 1;
	if (wsj_ep20c & (1 << .@s)) {
		mes "[" + .@name$ + "]";
		mes "Somebody's already there. Two of us would just get in each other's way.";
		close;
	}
	wsj_ep20c |= (1 << .@s);
	wsj_ep20b |= .@bit;
	setquest getarg(2);
	completequest getarg(2);
	mes "[" + .@name$ + "]";
	mes "Yes! I'll guard it with my life!";
	mes "Or at least until the next tide.";
	if ((wsj_ep20b & 992) == 992) {
		next;
		mes "^4d4dffAll five intermediate Rgans have somewhere to be. They wriggle off happily.^000000";
		completequest 8846;
		wsj_ep20b |= 8;
		callfunc "F_WSJ_Trust";
	}
	close;
}
