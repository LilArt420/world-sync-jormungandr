// Witnesses (690-709)
// ------------------------------------------------------------
function	script	F_WSJ_Witness	{
	.@n = getarg(0);
	if (wsj_ep21 != 690 || (wsj_ep21b & (1 << .@n))) {
		if (.@n == 0) { mes "[Exhausted Soldier]"; mes "I just want to sleep. Without dreaming."; }
		else if (.@n == 1) { mes "[Escaped Cultist]"; mes "I'm free. I keep telling myself that. I'm free."; }
		else { mes "[Wounded Soldier]"; mes "The healers say it will scar. Good. I want to remember."; }
		close;
	}
	if (.@n == 0) {
		mes "[Exhausted Soldier]";
		mes "The serpent? I saw it. At night, on the north shore of the Raised Land.";
		mes "Long as a ship. Grey-white, slick, like it had never seen the sun. It slid into the temple's shadow and was gone.";
		next;
		mes "[Exhausted Soldier]";
		mes "I haven't slept since. Every time I close my eyes, it's there.";
	} else if (.@n == 1) {
		mes "[Escaped Cultist]";
		mes "They call it the Firstborn. The bishops lead the chosen down to the deep halls, and the Firstborn... eats.";
		next;
		mes "[Escaped Cultist]";
		mes "I saw it take a man whole. He was singing. They're all singing when they go down.";
	} else {
		mes "[Wounded Soldier]";
		mes "Want proof? Look at my back.";
		next;
		mes "^4d4dffAcross the soldier's back runs a band of scars, each one shaped like a scale the size of a shield.^000000";
		next;
		mes "[Wounded Soldier]";
		mes "It brushed past me. Just brushed. I was lucky.";
	}
	setarray .@q, 18339, 18340, 18341;
	setquest .@q[.@n];
	completequest .@q[.@n];
	wsj_ep21b |= (1 << .@n);
	if ((wsj_ep21b & 7) == 7) {
		next;
		mes "^4d4dffThree witnesses. The serpent is real. Report to <NAVI>[Tris]<INFO>jor_mbase,204,193,0,101,0</INFO></NAVI>.^000000";
		callfunc "F_WSJ_Step21", 700, "18338", "18342";
	}
	close;
}

// Enters (or creates, then enters) a memorial dungeon for this character.
function	script	F_WSJ_EnterMd	{
	.@md$ = getarg(0);
	if (instance_enter(.@md$) == IE_OK)
		return;
	.@id = instance_create(.@md$, IM_CHAR);
	if (.@id < 0) {
		mes "^4d4dffThe way won't open right now. Try again in a moment.^000000";
		close;
	}
	if (instance_enter(.@md$) != IE_OK) {
		mes "^4d4dffThe way won't open right now. Try again in a moment.^000000";
		close;
	}
	return;
}
