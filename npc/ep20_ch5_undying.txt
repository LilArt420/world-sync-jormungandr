// The three tunnels. Each one turns out to be a dead end.
function	script	F_WSJ_Mark	{
	.@n = getarg(0);
	if (wsj_ep20 != 550 || (wsj_ep20b & (1 << .@n))) {
		mes "^4d4dffLehar's mark. The tunnel behind it is a dead end.^000000";
		close;
	}
	mes "^4d4dffAnother of Lehar's marks. Behind it, a narrow tunnel runs deep into the roots.^000000";
	next;
	progressbar "ffffff", 3;
	if (.@n == 0) {
		mes "^4d4dffAt the end of the tunnel: a pile of broken crates and the smell of machine oil.^000000";
		mes "^4d4dffSomeone stored things here, and moved them out in a hurry. No lab.^000000";
	} else if (.@n == 1) {
		mes "^4d4dffThe tunnel ends at a wall of fresh ice. On the other side, faint and blurred, you can see light.^000000";
		mes "^4d4dffThe ice is too thick to break. If there's a lab there, this isn't the way in.^000000";
	} else {
		mes "^4d4dffA nest of sleeping Rgan hatchlings. You back out very, very quietly.^000000";
		mes "^4d4dffNo lab here either.^000000";
	}
	setarray .@q, 18223, 18224, 18225;
	completequest .@q[.@n];
	wsj_ep20b |= (1 << .@n);
	if ((wsj_ep20b & 7) == 7) {
		next;
		mes "^4d4dffAll three tunnels, and no lab. Lehar won't be happy.";
		mes "Time to meet her at the <NAVI>[Sacred Roots]<INFO>jor_sanct,133,86,0,101,0</INFO></NAVI>.^000000";
		callfunc "F_WSJ_Step", 560, "18222", "18226";
	}
	close;
}
