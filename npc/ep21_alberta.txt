// The four messy spots of the port (step 325, bits 1-8).
function	script	F_WSJ_Mess	{
	.@n = getarg(0);
	if (wsj_ep21 != 325 || isbegin_quest(16805) == 1 || (wsj_ep21b & (1 << .@n))) {
		mes "^4d4dffClean enough, for a port.^000000";
		close;
	}
	mes "^4d4dffRotting nets, a broken crate and something that used to be a fish.^000000";
	next;
	if (select("Clean it up.:Leave it.") == 2)
		close;
	progressbar "ffffff", 3;
	setarray .@q, 16806, 16807, 16808, 16809;
	completequest .@q[.@n];
	wsj_ep21b |= (1 << .@n);
	mes "^4d4dffMuch better.^000000";
	if ((wsj_ep21b & 15) == 15) {
		mes "^4d4dffThat's all four. Back to <NAVI>[John]<INFO>jor_albe,239,115,0,101,0</INFO></NAVI>.^000000";
		callfunc "F_WSJ_Step21", 326, "", "16810";
	}
	close;
}

	killmonster instance_mapname("1@wtgs"), "All";

	end;
}
