// Moves Episode 21 on (as F_WSJ_Step does for Episode 20).
function	script	F_WSJ_Step21	{
	.@done$ = getarg(1);
	.@next$ = getarg(2);
	if (.@done$ != "") {
		explode .@d$, .@done$, ",";
		.@n = getarraysize(.@d$);
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = atoi(.@d$[.@i]);
			if (isbegin_quest(.@q) == 0)
				setquest .@q;
			if (isbegin_quest(.@q) == 1)
				completequest .@q;
		}
	}
	if (.@next$ != "") {
		explode .@x$, .@next$, ",";
		.@n = getarraysize(.@x$);
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = atoi(.@x$[.@i]);
			if (isbegin_quest(.@q) == 0)
				setquest .@q;
		}
	}
	wsj_ep21 = getarg(0);
	callfunc "F_WSJ_Scenes21";
	questinfo_refresh;
	return;
}

// The end of an Episode 21 chapter. Amounts aren't published: these
// follow Episode 20's, raised for level 230+, with Wigner Premium
// Exchange Tickets, the episode's currency.
function	script	F_WSJ_Chapter21	{
	getexp2 80000000,0;
	getexp 0,4000000;
	getitem 1001618,3;	// Ep21_Wigner_Ticket
	add_reputation_points( REPUTATION_EP19, 50 );
	return;
}
