// Hands in an Episode 21 daily.
// callfunc "F_WSJ_DailyDone21", <active quest>, <standby quest>, <base exp>, <tickets>;
function	script	F_WSJ_DailyDone21	{
	erasequest getarg(0);
	setquest getarg(1);
	getexp2 getarg(2), 0;
	getexp 0, getarg(2) / 15;
	if (getarg(3) > 0)
		getitem 1001618, getarg(3);	// Ep21_Wigner_Ticket
	return;
}

// A kill daily handed out by an Episode 21 NPC (as F_WSJ_DailyHunt).
// callfunc "F_WSJ_Hunt21", "<npc>", <active>, <standby>, "<what>", "<offer>", <exp>, <tickets>;
function	script	F_WSJ_Hunt21	{
	.@npc$ = getarg(0);
	.@q = getarg(1);
	.@w = getarg(2);
	switch (callfunc("F_WSJ_Daily", .@q, .@w)) {
	case 0:
		mes "[" + .@npc$ + "]";
		mes getarg(4);
		next;
		if (select("I'll do it.:Not today.") == 2)
			close;
		setquest .@q;
		mes "[" + .@npc$ + "]";
		mes "^FF0000" + getarg(3) + "^000000. Come back to me when it's done.";
		close;
	case 1:
		if (checkquest(.@q,HUNTING) != 2) {
			mes "[" + .@npc$ + "]";
			mes "Not done yet: ^FF0000" + getarg(3) + "^000000.";
			close;
		}
		mes "[" + .@npc$ + "]";
		mes "Done already? Good work. Here's your share.";
		callfunc "F_WSJ_DailyDone21", .@q, .@w, getarg(5), getarg(6);
		close;
	case 2:
		mes "[" + .@npc$ + "]";
		mes "That's enough for today. Come back after dawn.";
		close;
	}
	return;
}

// The applicants (24024-24029): five a day get interviewed.
function	script	F_WSJ_Applicant	{
	.@n = getarg(0);
	mes "[" + strnpcinfo(1) + "]";
	.@cur = 0;
	for (.@q = 24024; .@q <= 24028; .@q++)
		if (isbegin_quest(.@q) == 1) .@cur = .@q;
	if (!.@cur || (wsj_app & (1 << .@n))) {
		mes "I want to join Richard's mercenaries! Are you hiring?";
		close;
	}
	setarray .@line$, "I can carry two sacks of grain at once! Three if they're small!",
		"I used to be a cultist. I know where they keep the good snacks.",
		"I'm very good at running away. Is that a skill here?",
		"My grandmother fought in the first war. She says I'm too soft. I'm not!",
		"I can cook. Badly. But fast!",
		"I've got my own sword. It's a stick, but it's mine.";
	mes .@line$[.@n % 6];
	next;
	if (select("You'll do. Report to Valdaris.:Not yet. Train some more.") == 2) {
		mes "[" + strnpcinfo(1) + "]";
		mes "Aww.";
		close;
	}
	wsj_app |= (1 << .@n);
	completequest .@cur;
	erasequest .@cur;
	if (.@cur < 24028)
		setquest .@cur + 1;
	else
		setquest 24029;
	mes "[" + strnpcinfo(1) + "]";
	mes "Yes! YES! I won't let you down!";
	close;
}
