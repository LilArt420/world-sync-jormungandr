//===== World Sync: Jormungandr =============================
//= Episode 20: daily quests (reset at 04:00).
//= Official quest ids; rewards aren't published, so each gives
//= experience and Snake God Scales like the story's smaller steps.
//============================================================

// Daily state of one quest pair.
// callfunc("F_WSJ_Daily", <active quest>, <standby quest>)
// 0 = can be taken, 1 = in progress, 2 = done today (standby running).
function	script	F_WSJ_Daily	{
	.@q = getarg(0);
	.@w = getarg(1);
	if (isbegin_quest(.@q) == 1)
		return 1;
	if (isbegin_quest(.@w) == 1) {
		if (checkquest(.@w,PLAYTIME) == 2) {
			erasequest .@w;
			return 0;
		}
		return 2;
	}
	return 0;
}

// Hands in a daily: removes the active quest, starts the standby one,
// and gives the reward.
// callfunc "F_WSJ_DailyDone", <active quest>, <standby quest>, <base exp>, <scales>;
function	script	F_WSJ_DailyDone	{
	erasequest getarg(0);
	setquest getarg(1);
	getexp2 getarg(2), 0;
	getexp 0, getarg(2) / 15;
	if (getarg(3) > 0)
		getitem 1001251, getarg(3);	// EP20_D_Scale
	add_reputation_points( REPUTATION_EP19, 5 );
	return;
}

// Cleaning the Serpent's Maze (Supervisor, jor_sanct)
// ------------------------------------------------------------
jor_sanct,24,281,5	script	Cleaning Board#wsj	4_EP19_RGAN_R1,{
	if (!callfunc("F_WSJ_IsRgan", "Cleaner")) end;
	if (wsj_ep20 < 280) {
		mes "[Cleaner]";
		mes "The Supervisor hands out the cleaning. Not me. I just hold the mop.";
		close;
	}
	switch (callfunc("F_WSJ_Daily", 17710, 17711)) {
	case 0:
		mes "[Cleaner]";
		mes "The Supervisor says today's cleaning is the guards in the Serpent's Maze. Ten purple, ten yellow.";
		mes "He says you clean best. Want the job?";
		next;
		if (select("I'll do it.:Not today.") == 2)
			close;
		setquest 17710;
		mes "[Cleaner]";
		mes "^FF000010 Jormun Guard-P and 10 Jormun Guard-Y^000000 in the maze. Come back to me when you're done.";
		close;
	case 1:
		if (checkquest(17710,HUNTING) != 2) {
			mes "[Cleaner]";
			mes "Still dirty in there. Ten purple, ten yellow.";
			close;
		}
		mes "[Cleaner]";
		mes "Spotless! The Supervisor will be pleased. Here, from the cleaning budget.";
		callfunc "F_WSJ_DailyDone", 17710, 17711, 20000000, 1;
		close;
	case 2:
		mes "[Cleaner]";
		mes "That's enough cleaning for today. The maze gets dirty again overnight. Always does.";
		close;
	}
	end;

OnInit:
	questinfo( QTYPE_DAILYQUEST, QMARK_YELLOW, "wsj_ep20 >= 280 && isbegin_quest(17710) == 0 && (isbegin_quest(17711) == 0 || checkquest(17711,PLAYTIME) == 2)" );
	end;
}

// Boclinde's Extermination Operations (Voglinde, icas_in2)
// Pulling Out Jormungandr's Roots (Kororo, jor_twice)
// ------------------------------------------------------------
// These three give their dailies through the story NPCs' own scripts:
// see F_WSJ_DailyHunt below, called from Chichiring, Voglinde and Kororo.

// A kill daily handed out by a story NPC.
// callfunc "F_WSJ_DailyHunt", "<npc name>", <active>, <standby>, "<what to kill>", "<offer line>", <exp>;
function	script	F_WSJ_DailyHunt	{
	.@npc$ = getarg(0);
	.@q = getarg(1);
	.@w = getarg(2);
	switch (callfunc("F_WSJ_Daily", .@q, .@w)) {
	case 0:
		mes "[" + .@npc$ + "]";
		mes getarg(4);
		next;
		if (select("I'll take care of it.:Not today.") == 2)
			close;
		setquest .@q;
		mes "[" + .@npc$ + "]";
		mes "Defeat ^FF0000" + getarg(3) + "^000000 and come back to me.";
		close;
	case 1:
		if (checkquest(.@q,HUNTING) != 2) {
			mes "[" + .@npc$ + "]";
			mes "Not done yet: ^FF0000" + getarg(3) + "^000000.";
			close;
		}
		mes "[" + .@npc$ + "]";
		mes "Well done. That'll keep them quiet for a day. Take this for your trouble.";
		callfunc "F_WSJ_DailyDone", .@q, .@w, getarg(5), 1;
		close;
	case 2:
		mes "[" + .@npc$ + "]";
		mes "That's enough for today. Come back after dawn.";
		close;
	}
	return;
}
