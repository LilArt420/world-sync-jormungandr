// The Rgans' mana absorption devices. Each can be broken once a day;
// what's left of the core is Pure Magical Ore.
function	script	F_WSJ_Device	{
	.@n = getarg(0);
	if (wsj_ep20 < 100) {
		mes "^4d4dffA strange device driven into the ice, humming softly. Better not touch what you don't understand.^000000";
		close;
	}
	.@today = gettime(DT_YYYYMMDD);
	if (wsj_dev_day != .@today) {
		wsj_dev_day = .@today;
		wsj_dev = 0;
	}
	if (wsj_dev & (1 << .@n)) {
		mes "^4d4dffThe device is still smashed. The Rgans will be back to fix it tomorrow.^000000";
		close;
	}
	mes "^4d4dffA Rgan mana absorption device, draining the canyon. Its core glows with stolen mana.^000000";
	next;
	if (select("Smash it.:Leave it.") == 2)
		close;
	progressbar "ffffff", 3;
	specialeffect EF_LORD;
	wsj_dev |= (1 << .@n);
	mes "^4d4dffThe core cracks and the hum dies. The snow around it starts to sparkle again.^000000";
	if (rand(100) < 40)
		getitem 1001248,1;	// EP20_D_P_Ore
	close;
}
