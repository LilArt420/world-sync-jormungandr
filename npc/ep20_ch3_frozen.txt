// The three gossiping pairs. Overhearing one is enough for its rumour.
function	script	F_WSJ_Rumor	{
	.@n = getarg(0);
	if (!callfunc("F_WSJ_IsRgan", getarg(1))) end;
	if (wsj_ep20 != 500 || (wsj_ep20b & (1 << .@n))) {
		mes "[" + getarg(1) + "]";
		mes "Hm? Mind your own tail.";
		close;
	}
	if (.@n == 0) {
		mes "[Hoton Lokengand]";
		mes "Did you hear? Bishop Rasgand got thrown out of the maze. By a CAT.";
		next;
		mes "[Sohem Garyugand]";
		mes "Shh! Don't let him hear you. He's been smashing things for two tides.";
		mes "He says next time he'll go in with the heart shard, and no cat will stop him.";
	} else if (.@n == 1) {
		mes "[Misu Gozengand]";
		mes "The human Bagot's lab? It's on the 1st floor of the Roots. Behind one of the walls.";
		next;
		mes "[Naaryugand]";
		mes "Which wall?";
		next;
		mes "[Misu Gozengand]";
		mes "A wall. It's a lab. It's behind a wall. I'm not an architect.";
	} else {
		mes "[Kengundrgand]";
		mes "The cultists out on the floes have been chanting for days. Pulling the magic up through the ice.";
		next;
		mes "[Hanaungand]";
		mes "The Bishop says it doesn't matter if the maze gate fails. There's ^FF0000another way^000000 into the nest now.";
		mes "He's very pleased about it. I don't like it when he's pleased.";
	}
	setarray .@q, 23113, 23114, 23115;
	completequest .@q[.@n];
	wsj_ep20b |= (1 << .@n);
	if ((wsj_ep20b & 7) == 7) {
		next;
		mes "^4d4dffThat's enough rumours. Time to meet <NAVI>[Miriam]<INFO>jor_maze,150,33,0,101,0</INFO></NAVI> at the south end of the maze.^000000";
		callfunc "F_WSJ_Step", 510, "", "23117";
	}
	close;
}
