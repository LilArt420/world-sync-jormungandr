// The three searches. None of them gives anything useful.
function	script	F_WSJ_Search	{
	.@n = getarg(0);
	if (wsj_ep21 != 540 || (wsj_ep21b & (1 << .@n)))
		end;
	if (.@n == 0) {
		mes "[Believer]";
		mes "Escape? Who would want to escape? The Serpent loves us. The Serpent feeds us.";
		mes "...Please stop looking at me like that. Someone might be listening.";
	} else if (.@n == 1) {
		mes "[Cult Archivist]";
		mes "Ships? This is a temple, pilgrim, not a harbour. We keep the scriptures here.";
		mes "Now shoo. Your sandals are scuffing the floor.";
	} else {
		mes "[Stairway Guard]";
		mes "The upper floor is for the anointed. Where's your seal?";
		next;
		mes "[Stairway Guard]";
		mes "No seal, no stairs. Move along.";
	}
	setarray .@q, 17753, 17754, 17755;
	completequest .@q[.@n];
	wsj_ep21b |= (1 << .@n);
	if ((wsj_ep21b & 7) == 7) {
		next;
		mes "^4d4dffNothing. Time to meet the others at the <NAVI>[entrance]<INFO>jor_tmple1,188,72,0,101,0</INFO></NAVI>.^000000";
		callfunc "F_WSJ_Step21", 550, "", "17756";
	}
	close;
}
