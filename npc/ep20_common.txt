// Moves the story on: finishes the quests given, starts the next ones,
// sets the step and refreshes who is visible.
// callfunc "F_WSJ_Step", <new step>, "<quests to complete>", "<quests to start>";
// Quest lists are comma separated ids ("" for none).
function	script	F_WSJ_Step	{
	.@done$ = getarg(1);
	.@next$ = getarg(2);
	if (.@done$ != "") {
		explode .@d$, .@done$, ",";
		.@n = getarraysize(.@d$);
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = atoi(.@d$[.@i]);
			if (isbegin_quest(.@q) == 1)
				completequest .@q;
			else if (isbegin_quest(.@q) == 0) {
				setquest .@q;
				completequest .@q;
			}
		}
	}
	if (.@next$ != "") {
		explode .@x$, .@next$, ",";
		.@n = getarraysize(.@x$);
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = atoi(.@x$[.@i]);
			if (isbegin_quest(.@q) == 0)
				setquest .@q;
		}
	}
	wsj_ep20 = getarg(0);
	callfunc "F_WSJ_Scenes";
	questinfo_refresh;
	return;
}

// The end of a chapter: experience, Snake God Scales and Isgard
// reputation. The official amounts aren't published; these follow
// Episode 19's main story (getexp2 ~52M) for level 200+ players.
function	script	F_WSJ_Chapter	{
	getexp2 60000000,0;
	getexp 0,3500000;
	getitem 1001251,3;	// EP20_D_Scale
	add_reputation_points( REPUTATION_EP19, 50 );
	return;
}

// Rgans only talk to Rgans. Returns 1 when the player is transformed
// with a Rgan Transformation Scroll; otherwise the Rgan shoos them.
// callfunc("F_WSJ_IsRgan", "<npc display name>")
function	script	F_WSJ_IsRgan	{
	if (getstatus(SC_MONSTER_TRANSFORM,1) == 21530)
		return 1;
	mes "[" + getarg(0) + "]";
	mes "A... a human?!";
	mes "Go away, go away! Guards!";
	next;
	mes "^4d4dffYou'd better look like a Rgan first.";
	mes "Use a " + getitemname(101162) + ". Lunch, out on the Frozen Scale Glacier, makes them.^000000";
	close;
}

// Snakeberries come up from the hardened soil of the Serpent God's
// Roots, 2nd floor.
function	script	F_WSJ_Berry	{
	if (countitem(1001154) >= 10 && (wsj_ep20 < 200 || wsj_ep20 > 215) && wsj_ep20 != 360) {
		mes "There's hardened soil here. Snakeberries grow under it.";
		mes "But your pockets are full of them already.";
		close;
	}
	mes "Hardened soil, with a few red berries poking out.";
	mes "Dig some up?";
	next;
	if (select("Dig:Leave it") == 2)
		close;
	close2;
	progressbar "ffffff", 3;
	if (rand(100) < 70) {
		getitem 1001154,1;	// Ep20_Jor_Berry
		specialeffect2 EF_BUBBLE;
	} else
		dispbottom "Only roots and pebbles this time.";
	if (wsj_ep20 == 210 && countitem(1001154) >= 3) {
		callfunc "F_WSJ_Step", 215, "17698", "17719";
		dispbottom "That's three Snakeberries. Back to the Bishops at the Sacred Roots.";
	}
	end;
}

// Episode 20 Clear Ticket: the Episode Valkyrie (standart-npc) calls this
// to mark the main story finished, as the official ticket does. No
// rewards; the dailies open. $@wsj_ep20_skip tells her it's available
// (temporary, so it's gone when this mod is switched off).
function	script	F_WSJ_Skip20	{
	setarray .@q[0],
		16691, 16692, 16693, 16694, 16695, 16696, 16697, 16698, 16699, 16700,
		16701, 16702, 16703, 16704, 17690, 17691, 17692, 17693, 17694, 17695,
		17696, 17697, 17698, 17699, 17700, 17701, 17702, 17703, 17704, 17705,
		17706, 17707, 17708, 17709, 17712, 17713, 17714, 17715, 17716, 17719,
		8839, 8840, 8841, 8842, 8843, 8846, 8854,
		23100, 23101, 23102, 23103, 23104, 23105, 23106, 23107, 23108, 23109,
		23112, 23113, 23114, 23115, 23116, 23117, 23118,
		18220, 18221, 18222, 18223, 18224, 18225, 18226, 18227, 18228, 18229,
		18230, 18231, 18232;
	.@n = getarraysize(.@q);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (isbegin_quest(.@q[.@i]) == 1)
			erasequest .@q[.@i];
	}
	if (isbegin_quest(18233) == 0)
		setquest 18233;
	if (isbegin_quest(18233) == 1)
		completequest 18233;
	// The device from the canyon, taken back by the story.
	if (countitem(1001218))
		delitem 1001218, countitem(1001218);
	wsj_ep20 = 700;
	wsj_ep20b = 0;
	wsj_ep20c = 0;
	callfunc "F_WSJ_Scenes";
	return;
}

	end;
